function drawQR(svg){
  if(!svg) return;
  let cells='';
  const finder=(x,y)=>`<rect x="${x}" y="${y}" width="7" height="7" fill="#2b2924"/><rect x="${x+1}" y="${y+1}" width="5" height="5" fill="#fff"/><rect x="${x+2}" y="${y+2}" width="3" height="3" fill="#2b2924"/>`;
  for(let y=0;y<33;y++)for(let x=0;x<33;x++){
    if((x<9&&y<9)||(x>23&&y<9)||(x<9&&y>23)) continue;
    if(Math.abs(Math.sin(x*13.7+y*7.3+x*y*.31))>.52) cells+=`<rect x="${x}" y="${y}" width="1" height="1" fill="#2b2924"/>`;
  }
  svg.innerHTML = finder(0,0)+finder(26,0)+finder(0,26)+cells;
}
