let PG = [];

function renderBook(){
  PG = pages(); page = 0;
  $('#book-title').textContent = `${S.first} 님의 기억 여행`;
  $('#ship-to').textContent = `${S.name} 어르신`;
  $('#ship-cur').textContent = `3권 · ${CITY[S.city].sea}의 기억`;
  $('#ship-next').textContent = `4권 · ${CITY[S.city].market}의 하루`;
  $('#dots').innerHTML = PG.map((_,i)=>`<span class="dot${i===0?' on':''}"></span>`).join('');
  showPage(0);
}

function showPage(n){
  page = Math.max(0, Math.min(PG.length-1, n));
  const p = PG[page];
  const ABC = {'집중력':'#d95f4b','기억력':'#d95f4b','언어력':'#4b7dd9','시공간력':'#4b9dd9','지남력':'#7a5fd9','판단력':'#3f9271','정서':'#c9793f','세대연결':'#c9793f'};
  const ab = (p.tag && p.tag[3]) ? p.tag[3].map(a=>`<span class="abil" style="background:${ABC[a]||'#999'}">${a}</span>`).join('') : '';
  const study = p.tag ? `<div class="studybar"><span>📅 학습날짜 <span class="ln"></span>월 <span class="ln"></span>일</span><span>⏱ 학습시간 <span class="ln"></span>분</span></div>` : '';
  const day = p.tag ? `<div class="daynum">${p.tag[0]}<span>일 차</span></div>` : '';
  const tag = p.tag ? `${day}${study}<div style="margin-left:86px"><span class="ptag"><span class="num">${p.tag[0]}</span>${p.tag[1]}</span>${ab}<div class="psub">${p.tag[2]}</div></div>` : '';
  $('#pages').innerHTML = `<div class="page">${tag}${p.html}
    <div class="pfoot"><span>기억윤슬 · ${S.name} 님 전용판</span><span>${page+1} / ${PG.length}</span></div></div>`;
  $('#page-label').textContent = p.label;
  $$('#dots .dot').forEach((d,i)=>d.classList.toggle('on', i===page));
  $('#prev').disabled = page===0;
  $('#next').disabled = page===PG.length-1;
  initPage();
}
$('#prev').addEventListener('click',()=>showPage(page-1));
$('#next').addEventListener('click',()=>showPage(page+1));
